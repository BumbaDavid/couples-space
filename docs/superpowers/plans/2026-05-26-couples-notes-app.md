# Couples Notes App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the V1 Android Flutter app from
[docs/superpowers/specs/2026-05-26-couples-notes-app-design.md](../specs/2026-05-26-couples-notes-app-design.md):
two-account couples app with shared notes, per-user private notes, invite-code pairing,
per-user theme mode, and Firestore-rules-enforced isolation.

**Architecture:** Feature-first Flutter app. UI watches Riverpod `StreamProvider`s that
wrap repositories; repositories own all Firestore reads/writes. Pairing is two-phase:
the claiming user writes the couple + their own user doc atomically; the inviting user's
app backfills its own `coupleId` via a query the next time its auth gate runs. Security
is enforced entirely by Firestore rules, validated by emulator-based unit tests.

**Tech Stack:** Flutter (Android target), Firebase Auth (email/password), Cloud
Firestore, Riverpod, go_router, freezed, fake_cloud_firestore, `@firebase/rules-unit-testing`.

## Task 3: Firestore type converters

**Files:**

Firestore's Dart SDK round-trips `DateTime` ↔ `Timestamp` if you pass it through, but
`json_serializable` outputs an ISO string. We need a small adapter layer so models
encode to Firestore-native types. We do this once, centrally, rather than scattering it
through repositories.

import 'package:cloud_firestore/cloud_firestore.dart';

Map<String, dynamic> dateTimesToTimestamps(
  Map<String, dynamic> input,
  List<String> dateTimeFields,
) {
  final result = Map<String, dynamic>.from(input);
  for (final field in dateTimeFields) {
    final value = result[field];
    if (value is String) {
      result[field] = Timestamp.fromDate(DateTime.parse(value));
    }
  }
  return result;
}

Map<String, dynamic> timestampsToDateTimes(
  Map<String, dynamic> input,
  List<String> dateTimeFields,
) {
  final result = Map<String, dynamic>.from(input);
  for (final field in dateTimeFields) {
    final value = result[field];
    if (value is Timestamp) {
      result[field] = value.toDate().toUtc().toIso8601String();
    }
  }
  return result;
}
```

import 'package:cloud_firestore/cloud_firestore.dart';

import '../../../core/firebase/converters.dart';
import 'user_profile.dart';

class UserRepository {
  UserRepository(this._firestore);

  final FirebaseFirestore _firestore;
  static const _timeFields = ['createdAt'];

  CollectionReference<Map<String, dynamic>> get _col =>
      _firestore.collection('users');

  Stream<UserProfile?> watchUser(String userId) {
    return _col.doc(userId).snapshots().map(_decode);
  }

  Future<UserProfile?> getUser(String userId) async {
    final snap = await _col.doc(userId).get();
    return _decode(snap);
  }

  Future<void> createUser(UserProfile profile) async {
    final json = profile.toJson()..remove('id');
    final encoded = dateTimesToTimestamps(json, _timeFields);
    await _col.doc(profile.id).set(encoded);
  }

  Future<void> updateDisplayName(String userId, String displayName) {
    return _col.doc(userId).update({'displayName': displayName});
  }

  Future<void> updateThemeMode(String userId, AppThemeMode mode) {
    return _col.doc(userId).update({'themeMode': mode.name});
  }

  Future<void> setCoupleId(String userId, String coupleId) {
    return _col.doc(userId).update({'coupleId': coupleId});
  }

  UserProfile? _decode(DocumentSnapshot<Map<String, dynamic>> snap) {
    final data = snap.data();
    if (data == null) return null;
    final converted = timestampsToDateTimes(data, _timeFields);
    return UserProfile.fromJson({...converted, 'id': snap.id});
  }
}
```

import 'dart:math';

import 'package:cloud_firestore/cloud_firestore.dart';

import '../../../core/firebase/converters.dart';
import 'invite_code.dart';

class InviteCodeRepository {
  InviteCodeRepository(this._firestore, {Random? random})
      : _random = random ?? Random.secure();

  final FirebaseFirestore _firestore;
  final Random _random;
  static const _alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  static const _timeFields = ['createdAt', 'expiresAt'];

  CollectionReference<Map<String, dynamic>> get _col =>
      _firestore.collection('inviteCodes');

  String generateCode() {
    final buffer = StringBuffer();
    for (var i = 0; i < 6; i++) {
      buffer.write(_alphabet[_random.nextInt(_alphabet.length)]);
    }
    return buffer.toString();
  }

  Future<void> create(InviteCode invite) async {
    final json = invite.toJson()..remove('code');
    final encoded = dateTimesToTimestamps(json, _timeFields);
    await _col.doc(invite.code).set(encoded);
  }

  Future<InviteCode?> get(String code) async {
    final snap = await _col.doc(code).get();
    final data = snap.data();
    if (data == null) return null;
    final converted = timestampsToDateTimes(data, _timeFields);
    return InviteCode.fromJson({...converted, 'code': snap.id});
  }

  Future<void> delete(String code) {
    return _col.doc(code).delete();
  }
}
```

import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:uuid/uuid.dart';

import '../../../core/firebase/converters.dart';
import 'couple.dart';

class PairingException implements Exception {
  PairingException(this.reason);
  final String reason;
  @override
  String toString() => 'PairingException: $reason';
}

class CoupleRepository {
  CoupleRepository(this._firestore, {Uuid? uuid, DateTime Function()? now})
      : _uuid = uuid ?? const Uuid(),
        _now = now ?? DateTime.now;

  final FirebaseFirestore _firestore;
  final Uuid _uuid;
  final DateTime Function() _now;
  static const _timeFields = ['createdAt'];
  static const _inviteTimeFields = ['createdAt', 'expiresAt'];

  CollectionReference<Map<String, dynamic>> get _col =>
      _firestore.collection('couples');

  Stream<Couple?> watchCoupleById(String coupleId) {
    return _col.doc(coupleId).snapshots().map(_decode);
  }

  Future<Couple?> findCoupleForUser(String userId) async {
    final query = await _col
        .where('memberIds', arrayContains: userId)
        .limit(1)
        .get();
    if (query.docs.isEmpty) return null;
    return _decode(query.docs.first);
  }

  Future<String> claimInvite({
    required String code,
    required String currentUserId,
  }) async {
    final inviteRef = _firestore.collection('inviteCodes').doc(code);
    final inviteSnap = await inviteRef.get();
    final inviteData = inviteSnap.data();
    if (inviteData == null) {
      throw PairingException('Invite code not found');
    }
    final inviteJson = timestampsToDateTimes(inviteData, _inviteTimeFields);
    final createdBy = inviteJson['createdBy'] as String;
    final expiresAt = DateTime.parse(inviteJson['expiresAt'] as String);
    if (createdBy == currentUserId) {
      throw PairingException('Cannot claim your own invite');
    }
    if (expiresAt.isBefore(_now())) {
      throw PairingException('Invite code has expired');
    }

    final newCoupleId = _uuid.v4();
    final batch = _firestore.batch();
    batch.set(_col.doc(newCoupleId),
        dateTimesToTimestamps({
          'memberIds': [createdBy, currentUserId],
          'createdAt': _now().toUtc().toIso8601String(),
        }, _timeFields));
    batch.update(_firestore.collection('users').doc(currentUserId),
        {'coupleId': newCoupleId});
    batch.delete(inviteRef);
    await batch.commit();
    return newCoupleId;
  }

  Couple? _decode(DocumentSnapshot<Map<String, dynamic>> snap) {
    final data = snap.data();
    if (data == null) return null;
    final converted = timestampsToDateTimes(data, _timeFields);
    return Couple.fromJson({
      ...converted,
      'id': snap.id,
      'memberIds': List<String>.from(converted['memberIds'] as List),
    });
  }
}

import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../features/pairing/data/pairing_providers.dart';
import '../../features/profile/data/user_providers.dart';
import '../firebase/firebase_providers.dart';

enum AuthRoute { loading, signIn, pairing, home }

final authRouteProvider = Provider<AuthRoute>((ref) {
  final authAsync = ref.watch(authStateChangesProvider);
  final authUser = authAsync.asData?.value;
  if (authAsync.isLoading) return AuthRoute.loading;
  if (authUser == null) return AuthRoute.signIn;

  final profileAsync = ref.watch(currentUserProfileProvider);
  final profile = profileAsync.asData?.value;
  if (profileAsync.isLoading) return AuthRoute.loading;
  if (profile == null) return AuthRoute.loading;

  if (profile.coupleId != null) return AuthRoute.home;

  // Profile says no couple — consult the backfill query.
  final backfill = ref.watch(coupleBackfillProvider);
  return backfill.maybeWhen(
    data: (foundCoupleId) =>
        foundCoupleId == null ? AuthRoute.pairing : AuthRoute.home,
    orElse: () => AuthRoute.loading,
  );
});

final coupleBackfillProvider = FutureProvider<String?>((ref) async {
  final authUser = ref.watch(authStateChangesProvider).asData?.value;
  if (authUser == null) return null;
  final couples = ref.watch(coupleRepositoryProvider);
  final couple = await couples.findCoupleForUser(authUser.uid);
  if (couple == null) return null;
  final users = ref.watch(userRepositoryProvider);
  await users.setCoupleId(authUser.uid, couple.id);
  return couple.id;
});

class _NoteEditorScreenState extends ConsumerState<NoteEditorScreen> {
  late final TextEditingController _titleController;
  late final TextEditingController _bodyController;
  String? _persistedId;

  @override
  void initState() {
    super.initState();
    _titleController = TextEditingController(text: widget.existing?.title ?? '');
    _bodyController = TextEditingController(text: widget.existing?.body ?? '');

      }

- [ ] **Step 2: Write the rules**

Replace `firestore.rules` with:
```
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }
    function isSelf(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }
    function isCoupleMember(coupleId) {
      return isSignedIn() &&
        request.auth.uid in get(/databases/$(database)/documents/couples/$(coupleId)).data.memberIds;
    }

    match /users/{userId} {
      allow read: if isSelf(userId)
                  || (isSignedIn()
                      && resource.data.coupleId != null
                      && resource.data.coupleId == get(/databases/$(database)/documents/users/$(request.auth.uid)).data.coupleId);
      allow create: if isSelf(userId);
      allow update: if isSelf(userId);
      allow delete: if false;
    }

    match /couples/{coupleId} {
      allow read: if isCoupleMember(coupleId);
      allow create: if isSignedIn()
                    && request.auth.uid in request.resource.data.memberIds;
      allow update, delete: if false;

      match /sharedNotes/{noteId} {
        allow read, write: if isCoupleMember(coupleId);
      }
    }

    match /users/{userId}/privateNotes/{noteId} {
      allow read, write: if isSelf(userId);
    }

    match /inviteCodes/{code} {
      allow read: if isSignedIn();
      allow create: if isSignedIn()
                    && request.resource.data.createdBy == request.auth.uid;
      allow delete: if isSignedIn();
      allow update: if false;
    }
  }
}
